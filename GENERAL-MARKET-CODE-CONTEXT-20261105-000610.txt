type ParameterDefinition struct {
	Code     string
	Type     ParameterType
	Required bool
	Min      *float64
	Max      *float64
	Allowed  []string
}

type OfferConfiguration struct {
	ID           ID
	CategoryID   ID
	Version      uint64
	Parameters   []ParameterDefinition
	RequiredDocs []string
	Active       bool
	CreatedAt    time.Time
}

type OfferParameters map[string]any

func (c OfferConfiguration) Validate(values OfferParameters) error {
	if !validID(c.ID) || !validID(c.CategoryID) ||
		c.Version == 0 || !validTime(c.CreatedAt) {
		return ErrInvalidInput
	}

	for _, definition := range c.Parameters {
		value, exists := values[definition.Code]

		if !exists {
			if definition.Required {
				return fmt.Errorf("%w: missing parameter %s",
					ErrInvalidInput, definition.Code)
			}
			continue
		}

		switch definition.Type {
		case ParamString:
			if _, ok := value.(string); !ok {
				return fmt.Errorf("%w: parameter %s must be string",
					ErrInvalidInput, definition.Code)
			}

		case ParamBoolean:
			if _, ok := value.(bool); !ok {
				return fmt.Errorf("%w: parameter %s must be boolean",
					ErrInvalidInput, definition.Code)
			}

		case ParamNumber:
			number, ok := value.(float64)
			if !ok {
				return fmt.Errorf("%w: parameter %s must be number",
					ErrInvalidInput, definition.Code)
			}
			if definition.Min != nil && number < *definition.Min {
				return fmt.Errorf("%w: parameter %s below minimum",
					ErrInvalidInput, definition.Code)
			}
			if definition.Max != nil && number > *definition.Max {
				return fmt.Errorf("%w: parameter %s above maximum",
					ErrInvalidInput, definition.Code)
			}

		default:
			return ErrInvalidInput
		}

		if len(definition.Allowed) > 0 {
			text, ok := value.(string)
			if !ok {
				return ErrInvalidInput
			}

			allowed := false
			for _, candidate := range definition.Allowed {
				if candidate == text {
					allowed = true
					break
				}
			}
			if !allowed {
				return fmt.Errorf("%w: parameter %s is not allowed",
					ErrInvalidInput, definition.Code)
			}
		}
	}

	return nil
}

type QualificationState string

const (
	QualificationPending  QualificationState = "pending"
	QualificationApproved QualificationState = "approved"
	QualificationRejected QualificationState = "rejected"
	QualificationExpired  QualificationState = "expired"
)

type CompanyQualification struct {
	CompanyID    ID
	CategoryID   ID
	ActivityCode string
	State        QualificationState
	Documents    []string
	ValidUntil   time.Time
	Version      uint64
}

func (q CompanyQualification) IsEligible(
	categoryID ID,
	activityCode string,
	requiredDocuments []string,
	now time.Time,
) error {
	if !validID(q.CompanyID) || !validID(categoryID) ||
		!validTime(now) || q.Version == 0 {
		return ErrInvalidInput
	}

	if q.State != QualificationApproved ||
		q.CategoryID != categoryID ||
		q.ActivityCode != activityCode {
		return ErrNotEligible
	}

	if !q.ValidUntil.IsZero() && now.After(q.ValidUntil) {
		return ErrNotEligible
	}

	available := make(map[string]bool, len(q.Documents))
	for _, document := range q.Documents {
		available[document] = true
	}

	for _, document := range requiredDocuments {
		if !available[document] {
			return fmt.Errorf("%w: missing document %s",
				ErrNotEligible, document)
		}
	}

	return nil
}

type Offer struct {
	ID                    ID

	OwnerID               ID

	Title                 string

func (o *Offer) Validate() error {
	if o == nil || !validID(o.ID) || !validID(o.MarketID) ||
		!validID(o.OwnerID) || !validID(o.PublicCategoryID) ||
		!validID(o.SpecializedCategoryID) ||
		!validID(o.ConfigurationID) ||
		o.ConfigurationVersion == 0 ||
		strings.TrimSpace(o.Title) == "" ||
		o.Quantity <= 0 || o.Remaining < 0 ||
		o.Remaining > o.Quantity || o.Price < 0 ||
		strings.TrimSpace(o.Unit) == "" ||
		strings.TrimSpace(o.Currency) == "" ||
		o.Version == 0 || !validTime(o.CreatedAt) ||
		!validTime(o.UpdatedAt) {
		return ErrInvalidInput
	}
	return nil
}

func (o *Offer) SubmitForReview(now time.Time) error {
	if err := o.Validate(); err != nil {
		return err
	}
	if o.State != OfferDraft || !validTime(now) {
		return ErrInvalidTransition
	}
	o.State = OfferReview
	o.touch(now)
	return nil
}

func (o *Offer) Approve(now time.Time) error {
	if o.State != OfferReview || !validTime(now) {
		return ErrInvalidTransition
	}
	o.State = OfferApproved
	o.touch(now)
	return nil
}

func (o *Offer) Publish(now time.Time) error {
	if o.State != OfferApproved || !validTime(now) {
		return ErrInvalidTransition
	}
	if o.Remaining <= 0 {
		return ErrNotEligible
	}
	if !o.ValidUntil.IsZero() && !now.Before(o.ValidUntil) {
		return ErrNotEligible
	}
	o.State = OfferPublished
	o.touch(now)
	return nil
}

func (o *Offer) Suspend(now time.Time) error {
	if o.State != OfferPublished || !validTime(now) {
		return ErrInvalidTransition
	}
	o.State = OfferSuspended
	o.touch(now)
	return nil
}

func (o *Offer) Close(now time.Time) error {
	if o.State == OfferClosed || !validTime(now) {
		return ErrInvalidTransition
	}
	o.State = OfferClosed
	o.touch(now)
	return nil
}

func (o *Offer) touch(now time.Time) {
	o.UpdatedAt = now.UTC()
	o.Version++
}

		!validID(r.CategoryID) || r.Quantity <= 0 ||

		!validTime(r.CreatedAt) {

func (p SupplyProposal) IsCompatible(r SupplyRequest) bool {
	return p.RequestID == r.ID &&
		p.Quantity >= r.Quantity &&
		p.UnitPrice > 0 &&
		p.Currency == r.Currency &&
		p.State == ProposalSubmitted
}

func SelectProposal(
	request *SupplyRequest,
	proposal SupplyProposal,
) error {
	if request == nil || request.State != RequestOpen {
		return ErrInvalidTransition
	}
	if !validID(proposal.ID) || !proposal.IsCompatible(*request) {
		return ErrNotEligible
	}
	if request.MaxPrice > 0 && proposal.UnitPrice > request.MaxPrice {
		return ErrNotEligible
	}

	request.SelectedProposalID = proposal.ID
	request.State = RequestSelected
	return nil
}

		!validID(o.MarketID) || !validID(o.UserID) ||
		(o.Side != Buy && o.Side != Sell) ||
		o.Price <= 0 || o.Quantity <= 0 ||
		o.Filled < 0 || o.Filled > o.Quantity ||

	}

func (o *Order) Open() error {
	if err := o.Validate(); err != nil {
		return err
	}
	if o.State != OrderPending {
		return ErrInvalidTransition
	}
	o.State = OrderOpen
	o.Version++
	return nil
}

func (o *Order) Cancel() error {
	if o.State != OrderOpen && o.State != OrderPartial {
		return ErrInvalidTransition
	}
	o.State = OrderCancelled
	o.Version++
	return nil
}

func (o *Order) Remaining() int64 {
	return o.Quantity - o.Filled
}

type MatchResult struct {
	BuyOrderID  ID
	SellOrderID ID
	Quantity    int64
	Price       int64
}

func MatchOrders(buy, sell *Order) (MatchResult, error) {
	if buy == nil || sell == nil {
		return MatchResult{}, ErrInvalidInput
	}
	if err := buy.Validate(); err != nil {
		return MatchResult{}, err
	}
	if err := sell.Validate(); err != nil {
		return MatchResult{}, err
	}

	if buy.Side != Buy || sell.Side != Sell ||
		buy.State != OrderOpen && buy.State != OrderPartial ||
		sell.State != OrderOpen && sell.State != OrderPartial {
		return MatchResult{}, ErrInvalidTransition
	}

	if buy.MarketID != sell.MarketID ||
		buy.OfferID != sell.OfferID ||
		buy.UserID == sell.UserID ||
		buy.Price < sell.Price {
		return MatchResult{}, ErrNotEligible
	}

	quantity := buy.Remaining()
	if sell.Remaining() < quantity {
		quantity = sell.Remaining()
	}
	if quantity <= 0 {
		return MatchResult{}, ErrNotEligible
	}

	// Price-time rule: older order is the maker.
	executionPrice := buy.Price
	if sell.CreatedAt.Before(buy.CreatedAt) ||
		sell.CreatedAt.Equal(buy.CreatedAt) {
		executionPrice = sell.Price
	}

	buy.Filled += quantity
	sell.Filled += quantity

	if buy.Remaining() == 0 {
		buy.State = OrderFilled
	} else {
		buy.State = OrderPartial
	}

	if sell.Remaining() == 0 {
		sell.State = OrderFilled
	} else {
		sell.State = OrderPartial
	}

	buy.Version++
	sell.Version++

	return MatchResult{
		BuyOrderID:  buy.ID,
		SellOrderID: sell.ID,
		Quantity:    quantity,
		Price:       executionPrice,
	}, nil
}

func SortOrderBook(orders []Order) {
	sort.SliceStable(orders, func(i, j int) bool {
		a, b := orders[i], orders[j]

		if a.Side != b.Side {
			return a.Side == Buy
		}
		if a.Price != b.Price {
			if a.Side == Buy {
				return a.Price > b.Price
			}
			return a.Price < b.Price
		}
		return a.CreatedAt.Before(b.CreatedAt)
	})
}

		!validID(t.SellerID) || !validID(t.OfferID) ||

func (t *Trade) RecalculateSettlementReadiness() {
	if t.Contract == ContractFailed ||
		t.Services == ServiceFailed ||
		t.Delivery == DeliveryFailed {
		t.SettlementReadiness = SettlementNotReady
		return
	}

	contractOK := t.Contract == ContractNotRequired ||
		t.Contract == ContractCreated
	servicesOK := t.Services == ServiceNotRequired ||
		t.Services == ServiceCompleted
	deliveryOK := t.Delivery == DeliveryNotRequired ||
		t.Delivery == DeliveryConfirmed

	if contractOK && servicesOK && deliveryOK {
		t.SettlementReadiness = SettlementReady
	} else {
		t.SettlementReadiness = SettlementNotReady
	}
}

type StepRequirement string

const (
	StepRequired    StepRequirement = "required"
	StepOptional    StepRequirement = "optional"
	StepConditional StepRequirement = "conditional"
)

type ExecutionStep struct {
	Code              string
	Order             int
	Requirement       StepRequirement
	RequiredDocuments []string
	DependsOn         []string
}

type ExecutionPolicy struct {
	ID              ID
	Name            string
	Version         uint64
	OfferCategoryID ID
	Steps           []ExecutionStep
	Active          bool
	CreatedAt       time.Time
}

func (p ExecutionPolicy) Validate() error {
	if !validID(p.ID) || !validID(p.OfferCategoryID) ||
		strings.TrimSpace(p.Name) == "" ||
		p.Version == 0 || !validTime(p.CreatedAt) ||
		len(p.Steps) == 0 {
		return ErrInvalidInput
	}

	seen := make(map[string]bool)
	for _, step := range p.Steps {
		if strings.TrimSpace(step.Code) == "" ||
			step.Order < 1 || seen[step.Code] {
			return ErrInvalidInput
		}
		if step.Requirement != StepRequired &&
			step.Requirement != StepOptional &&
			step.Requirement != StepConditional {
			return ErrInvalidInput
		}
		seen[step.Code] = true
	}

	for _, step := range p.Steps {
		for _, dependency := range step.DependsOn {
			if !seen[dependency] || dependency == step.Code {
				return ErrPolicyViolation
			}
		}
	}

	return nil
}

type ExecutionProgress struct {
	TradeID        ID
	PolicyID       ID
	PolicyVersion  uint64
	CompletedSteps map[string]bool
}

func (p ExecutionProgress) CanComplete(
	policy ExecutionPolicy,
	stepCode string,
) error {
	if p.PolicyID != policy.ID ||
		p.PolicyVersion != policy.Version {
		return ErrConflict
	}

	var selected *ExecutionStep
	for i := range policy.Steps {
		if policy.Steps[i].Code == stepCode {
			selected = &policy.Steps[i]
			break
		}
	}
	if selected == nil {
		return ErrNotFound
	}

	for _, dependency := range selected.DependsOn {
		if !p.CompletedSteps[dependency] {
			return fmt.Errorf("%w: dependency %s",
				ErrPolicyViolation, dependency)
		}
	}
	return nil
}

type CommodityLot struct {
	ID                 ID
	OfferID            ID
	CommodityCode      string
	Quantity           int64
	Unit               string
	Verified           bool
	AggregationAllowed bool
}

type CommodityAggregation struct {
	ID            ID
	CommodityCode string
	Unit          string
	LotIDs        []ID
	TotalQuantity int64
	CreatedAt     time.Time
}

func AggregateCommodity(
	id ID,
	lots []CommodityLot,
	minimumQuantity int64,
	now time.Time,
) (CommodityAggregation, error) {
	if !validID(id) || len(lots) == 0 ||
		minimumQuantity <= 0 || !validTime(now) {
		return CommodityAggregation{}, ErrInvalidInput
	}

	first := lots[0]
	if first.CommodityCode == "" || first.Unit == "" {
		return CommodityAggregation{}, ErrInvalidInput
	}

	result := CommodityAggregation{
		ID:            id,
		CommodityCode: first.CommodityCode,
		Unit:          first.Unit,
		CreatedAt:     now.UTC(),
	}

	seen := make(map[ID]bool)

	for _, lot := range lots {
		if !validID(lot.ID) || lot.Quantity <= 0 ||
			lot.CommodityCode != first.CommodityCode ||
			lot.Unit != first.Unit ||
			!lot.Verified || !lot.AggregationAllowed ||
			seen[lot.ID] {
			return CommodityAggregation{}, ErrNotEligible
		}

		seen[lot.ID] = true
		result.LotIDs = append(result.LotIDs, lot.ID)
		result.TotalQuantity += lot.Quantity
	}

	if result.TotalQuantity < minimumQuantity {
		return CommodityAggregation{}, ErrNotEligible
	}

	return result, nil
}

	ExternalSent      ExternalRequestState = "sent"

	ExternalFailed    ExternalRequestState = "failed"
)

type ExternalRequest struct {
	ID                 ID
	CorrelationID      ID
	IdempotencyKey     string
	InstitutionCode    string
	Purpose            string
	Scope              []string
	PayloadReference   string
	State              ExternalRequestState
	ExternalTrackingID string
	ResponseReference  string
	Attempt            int
	CreatedAt          time.Time
	UpdatedAt          time.Time
}

func (r ExternalRequest) ValidateAuthorization(
	grantedScopes []string,
) error {
	if !validID(r.ID) || !validID(r.CorrelationID) ||
		strings.TrimSpace(r.IdempotencyKey) == "" ||
		strings.TrimSpace(r.InstitutionCode) == "" ||
		strings.TrimSpace(r.Purpose) == "" ||
		len(r.Scope) == 0 || !validTime(r.CreatedAt) {
		return ErrInvalidInput
	}

	allowed := make(map[string]bool)
	for _, scope := range grantedScopes {
		allowed[scope] = true
	}

	for _, scope := range r.Scope {
		if !allowed[scope] {
			return ErrExternalNotAllowed
		}
	}
	return nil
}

package domain

import (
	"errors"
	"fmt"
	"strings"
	"time"

var (
	ErrInvalidMarketID       = errors.New("invalid market id")
	ErrInvalidMarketCode     = errors.New("invalid market code")
	ErrInvalidMarketName     = errors.New("invalid market name")
	ErrInvalidMarketTimezone = errors.New("invalid market timezone")
	ErrInvalidTimestamp      = errors.New("invalid timestamp")
	ErrInvalidMarketState    = errors.New("invalid market state")
	ErrMarketClosed          = errors.New("market is closed")
)

type MarketID string

type MarketStatus string

const (
	MarketStatusDraft     MarketStatus = "draft"
	MarketStatusActive    MarketStatus = "active"
	MarketStatusSuspended MarketStatus = "suspended"
	MarketStatusClosed    MarketStatus = "closed"
)

func (s MarketStatus) IsValid() bool {
	switch s {
	case MarketStatusDraft,
		MarketStatusActive,
		MarketStatusSuspended,
		MarketStatusClosed:
		return true
	default:
		return false
	}
}

type CreateMarketInput struct {
	ID          MarketID
	Code        string
	Name        string
	Description string
	Timezone    string
}

type Market struct {
	id          MarketID
	code        string
	name        string
	description string
	timezone    string

	status    MarketStatus
	version   uint64
	createdAt time.Time
	updatedAt time.Time
}

func NewMarket(input CreateMarketInput, now time.Time) (*Market, error) {
	if strings.TrimSpace(string(input.ID)) == "" {
		return nil, ErrInvalidMarketID
	}

	code := normalizeMarketCode(input.Code)
	if !isValidMarketCode(code) {
		return nil, ErrInvalidMarketCode
	}

	name := strings.TrimSpace(input.Name)
	if name == "" || len([]rune(name)) > 160 {
		return nil, ErrInvalidMarketName
	}

	if now.IsZero() {
		return nil, ErrInvalidTimestamp
	}

	timezone := strings.TrimSpace(input.Timezone)
	if timezone == "" {
		timezone = "UTC"
	}

	if _, err := time.LoadLocation(timezone); err != nil {
		return nil, fmt.Errorf("%w: %s", ErrInvalidMarketTimezone, timezone)
	}

	return &Market{
		id:          input.ID,
		code:        code,
		name:        name,
		description: strings.TrimSpace(input.Description),
		timezone:    timezone,
		status:      MarketStatusDraft,
		version:     1,
		createdAt:   now.UTC(),
		updatedAt:   now.UTC(),
	}, nil
}

func normalizeMarketCode(value string) string {
	return strings.ToUpper(strings.TrimSpace(value))
}

func isValidMarketCode(value string) bool {
	if len(value) < 2 || len(value) > 32 {
		return false
	}

	for _, r := range value {
		valid := (r >= 'A' && r <= 'Z') ||
			(r >= '0' && r <= '9') ||
			r == '-' ||
			r == '_'

		if !valid {
			return false
		}
	}

	return true
}

func (m *Market) ID() MarketID {
	return m.id
}

func (m *Market) Code() string {
	return m.code
}

func (m *Market) Name() string {
	return m.name
}

func (m *Market) Description() string {
	return m.description
}

func (m *Market) Timezone() string {
	return m.timezone
}

func (m *Market) Status() MarketStatus {
	return m.status
}

func (m *Market) Version() uint64 {
	return m.version
}

func (m *Market) CreatedAt() time.Time {
	return m.createdAt
}

func (m *Market) UpdatedAt() time.Time {
	return m.updatedAt
}

func (m *Market) UpdateDetails(
	name string,
	description string,
	timezone string,
	now time.Time,
) error {
	if m.status == MarketStatusClosed {
		return ErrMarketClosed
	}

	name = strings.TrimSpace(name)
	if name == "" || len([]rune(name)) > 160 {
		return ErrInvalidMarketName
	}

	timezone = strings.TrimSpace(timezone)
	if timezone == "" {
		timezone = "UTC"
	}

	if _, err := time.LoadLocation(timezone); err != nil {
		return fmt.Errorf("%w: %s", ErrInvalidMarketTimezone, timezone)
	}

	if err := m.validateTimestamp(now); err != nil {
		return err
	}

	m.name = name
	m.description = strings.TrimSpace(description)
	m.timezone = timezone
	m.touch(now)

	return nil
}

func (m *Market) Activate(now time.Time) error {
	switch m.status {
	case MarketStatusDraft, MarketStatusSuspended:
		if err := m.validateTimestamp(now); err != nil {
			return err
		}

		m.status = MarketStatusActive
		m.touch(now)
		return nil

	case MarketStatusActive:
		return ErrInvalidMarketState

	case MarketStatusClosed:
		return ErrMarketClosed

	default:
		return ErrInvalidMarketState
	}
}

func (m *Market) Suspend(now time.Time) error {
	if m.status != MarketStatusActive {
		return ErrInvalidMarketState
	}

	if err := m.validateTimestamp(now); err != nil {
		return err
	}

	m.status = MarketStatusSuspended
	m.touch(now)

	return nil
}

func (m *Market) Close(now time.Time) error {
	if m.status == MarketStatusClosed {
		return ErrInvalidMarketState
	}

	if err := m.validateTimestamp(now); err != nil {
		return err
	}

	m.status = MarketStatusClosed
	m.touch(now)

	return nil
}

func (m *Market) IsOperational() bool {
	return m.status == MarketStatusActive
}

func (m *Market) validateTimestamp(now time.Time) error {
	if now.IsZero() {
		return ErrInvalidTimestamp
	}

	if now.Before(m.updatedAt) {
		return ErrInvalidTimestamp
	}

	return nil
}

func (m *Market) touch(now time.Time) {
	m.updatedAt = now.UTC()
	m.version++
}

	name := strings.TrimSpace(input.Name)
	if name == "" || len([]rune(name)) > 160 {
		return nil, ErrInvalidCategoryName
	}

	if !input.Kind.IsValid() {
		return nil, ErrInvalidCategoryKind
	}

	if input.ParentID != "" && input.ParentID == input.ID {
		return nil, ErrInvalidParentCategory
	}

	if now.IsZero() {
		return nil, ErrInvalidTimestamp
	}

	return &Category{
		id:        input.ID,
		code:      code,
		name:      name,
		kind:      input.Kind,
		parentID:  input.ParentID,
		sortOrder: input.SortOrder,
		status:    CategoryStatusDraft,
		version:   1,
		createdAt: now.UTC(),
		updatedAt: now.UTC(),
	}, nil
}

func normalizeCategoryCode(value string) string {
	return strings.ToUpper(strings.TrimSpace(value))
}

func isValidCategoryCode(value string) bool {
	if len(value) < 2 || len(value) > 64 {
		return false
	}

	for _, r := range value {
		valid := (r >= 'A' && r <= 'Z') ||
			(r >= '0' && r <= '9') ||
			r == '-' ||
			r == '_' ||
			r == '.'

		if !valid {
			return false
		}
	}

	return true
}

func (c *Category) Rename(name string, now time.Time) error {
	if c.status == CategoryStatusArchived {
		return ErrCategoryArchived
	}

	name = strings.TrimSpace(name)
	if name == "" || len([]rune(name)) > 160 {
		return ErrInvalidCategoryName
	}

	if err := c.validateTimestamp(now); err != nil {
		return err
	}

	if name == c.name {
		return nil
	}

	c.name = name
	c.touch(now)

	return nil
}

func (c *Category) ChangeParent(parentID CategoryID, now time.Time) error {
	if c.status == CategoryStatusArchived {
		return ErrCategoryArchived
	}

	if parentID != "" && parentID == c.id {
		return ErrInvalidParentCategory
	}

	if err := c.validateTimestamp(now); err != nil {
		return err
	}

	if parentID == c.parentID {
		return nil
	}

	c.parentID = parentID
	c.touch(now)

	return nil
}

func (c *Category) ChangeSortOrder(sortOrder int, now time.Time) error {
	if c.status == CategoryStatusArchived {
		return ErrCategoryArchived
	}

	if err := c.validateTimestamp(now); err != nil {
		return err
	}

	if sortOrder == c.sortOrder {
		return nil
	}

	c.sortOrder = sortOrder
	c.touch(now)

	return nil
}

func (c *Category) Activate(now time.Time) error {
	switch c.status {
	case CategoryStatusDraft:
		if err := c.validateTimestamp(now); err != nil {
			return err
		}

		c.status = CategoryStatusActive
		c.touch(now)
		return nil

	case CategoryStatusActive:
		return ErrInvalidCategoryMutation

	case CategoryStatusArchived:
		return ErrCategoryArchived

	default:
		return ErrInvalidCategoryStatus
	}
}

func (c *Category) Archive(now time.Time) error {
	if c.status == CategoryStatusArchived {
		return ErrInvalidCategoryMutation
	}

	if err := c.validateTimestamp(now); err != nil {
		return err
	}

	c.status = CategoryStatusArchived
	c.touch(now)

	return nil
}

func (c *Category) IsSelectable() bool {
	return c.status == CategoryStatusActive
}

func (c *Category) validateTimestamp(now time.Time) error {
	if now.IsZero() || now.Before(c.updatedAt) {
		return ErrInvalidTimestamp
	}

	return nil
}

func (c *Category) touch(now time.Time) {
	c.updatedAt = now.UTC()
	c.version++
}

var (
	ErrInvalidCategoryMappingID = errors.New("invalid category mapping id")
	ErrInvalidCategoryMapping   = errors.New("invalid category mapping")
	ErrCategoryMappingArchived  = errors.New("category mapping is archived")
	ErrCategoryMappingState     = errors.New("invalid category mapping state")
	ErrCategoryKindsMismatch    = errors.New("category kinds do not match mapping direction")
	ErrCategoryNotActive        = errors.New("both categories must be active")
)

type CategoryMappingID string

type CategoryMappingStatus string

const (
	CategoryMappingStatusDraft    CategoryMappingStatus = "draft"
	CategoryMappingStatusActive   CategoryMappingStatus = "active"
	CategoryMappingStatusArchived CategoryMappingStatus = "archived"
)

type CreateCategoryMappingInput struct {
	ID                    CategoryMappingID
	PublicCategoryID      CategoryID
	SpecializedCategoryID CategoryID
	MarketID              string
	Description           string
}

type CategoryMapping struct {
	id                    CategoryMappingID
	publicCategoryID      CategoryID
	specializedCategoryID CategoryID
	marketID              string
	description           string

	status    CategoryMappingStatus
	version   uint64
	createdAt time.Time
	updatedAt time.Time
}

func NewCategoryMapping(
	input CreateCategoryMappingInput,
	now time.Time,
) (*CategoryMapping, error) {
	if strings.TrimSpace(string(input.ID)) == "" {
		return nil, ErrInvalidCategoryMappingID
	}

	if strings.TrimSpace(string(input.PublicCategoryID)) == "" ||
		strings.TrimSpace(string(input.SpecializedCategoryID)) == "" ||
		input.PublicCategoryID == input.SpecializedCategoryID {
		return nil, ErrInvalidCategoryMapping
	}

	if now.IsZero() {
		return nil, ErrInvalidTimestamp
	}

	return &CategoryMapping{
		id:                    input.ID,
		publicCategoryID:      input.PublicCategoryID,
		specializedCategoryID: input.SpecializedCategoryID,
		marketID:              strings.TrimSpace(input.MarketID),
		description:           strings.TrimSpace(input.Description),
		status:                CategoryMappingStatusDraft,
		version:               1,
		createdAt:             now.UTC(),
		updatedAt:             now.UTC(),
	}, nil
}

func (m *CategoryMapping) ID() CategoryMappingID {
	return m.id
}

func (m *CategoryMapping) PublicCategoryID() CategoryID {
	return m.publicCategoryID
}

func (m *CategoryMapping) SpecializedCategoryID() CategoryID {
	return m.specializedCategoryID
}

func (m *CategoryMapping) MarketID() string {
	return m.marketID
}

func (m *CategoryMapping) Description() string {
	return m.description
}

func (m *CategoryMapping) Status() CategoryMappingStatus {
	return m.status
}

func (m *CategoryMapping) Version() uint64 {
	return m.version
}

func (m *CategoryMapping) CreatedAt() time.Time {
	return m.createdAt
}

func (m *CategoryMapping) UpdatedAt() time.Time {
	return m.updatedAt
}

func (m *CategoryMapping) Activate(
	publicCategory *Category,
	specializedCategory *Category,
	now time.Time,
) error {
	if m.status == CategoryMappingStatusArchived {
		return ErrCategoryMappingArchived
	}

	if m.status != CategoryMappingStatusDraft {
		return ErrCategoryMappingState
	}

	if publicCategory == nil || specializedCategory == nil {
		return ErrInvalidCategoryMapping
	}

	if publicCategory.ID() != m.publicCategoryID ||
		specializedCategory.ID() != m.specializedCategoryID {
		return ErrInvalidCategoryMapping
	}

	if publicCategory.Kind() != CategoryKindGeneral ||
		specializedCategory.Kind() != CategoryKindSpecialized {
		return ErrCategoryKindsMismatch
	}

	if !publicCategory.IsSelectable() || !specializedCategory.IsSelectable() {
		return ErrCategoryNotActive
	}

	if now.IsZero() || now.Before(m.updatedAt) {
		return ErrInvalidTimestamp
	}

	m.status = CategoryMappingStatusActive
	m.touch(now)

	return nil
}

func (m *CategoryMapping) Archive(now time.Time) error {
	if m.status == CategoryMappingStatusArchived {
		return ErrCategoryMappingState
	}

	if now.IsZero() || now.Before(m.updatedAt) {
		return ErrInvalidTimestamp
	}

	m.status = CategoryMappingStatusArchived
	m.touch(now)

	return nil
}

func (m *CategoryMapping) IsUsable() bool {
	return m.status == CategoryMappingStatusActive
}

func (m *CategoryMapping) touch(now time.Time) {
	m.updatedAt = now.UTC()
	m.version++
}

func (h Handler) Register(mux *http.ServeMux) {

	mux.HandleFunc("GET /api/v1/catalogue/offers/{id}", h.GetOffer)

func (h Handler) ListOffers(w http.ResponseWriter, r *http.Request) {
	page, err := parsePositiveInt(r.URL.Query().Get("page"), 1, 1000000)
	if err != nil {
		writeError(w, http.StatusBadRequest, "invalid_page", "page must be a positive integer.")
		return
	}

	pageSize, err := parsePositiveInt(r.URL.Query().Get("page_size"), defaultPageSize, maxPageSize)
	if err != nil {
		writeError(w, http.StatusBadRequest, "invalid_page_size", "page_size must be between 1 and 100.")
		return
	}

	search := strings.TrimSpace(r.URL.Query().Get("q"))
	if len(search) > maxSearchLength {
		writeError(w, http.StatusBadRequest, "search_too_long", "q must not exceed 120 characters.")
		return
	}

	categoryID := strings.TrimSpace(r.URL.Query().Get("category_id"))
	marketID := strings.TrimSpace(r.URL.Query().Get("market_id"))

	if h.DB == nil {
		writeError(w, http.StatusServiceUnavailable, "database_unavailable", "Catalogue storage is unavailable.")
		return
	}

	const filter = `
		FROM offers o
		INNER JOIN markets m ON m.id = o.market_id
		LEFT JOIN public_categories pc ON pc.id = o.public_category_id
		WHERE o.status = 'published'
		  AND m.status = 'active'
		  AND (o.valid_from IS NULL OR o.valid_from <= now())
		  AND (o.valid_until IS NULL OR o.valid_until > now())
		  AND ($1 = '' OR o.title ILIKE '%' || $1 || '%' OR COALESCE(o.summary, '') ILIKE '%' || $1 || '%')
		  AND ($2 = '' OR o.public_category_id::text = $2)
		  AND ($3 = '' OR o.market_id::text = $3)
	`

	var total int64
	countQuery := "SELECT count(*) " + filter
	if err := h.DB.QueryRowContext(r.Context(), countQuery, search, categoryID, marketID).Scan(&total); err != nil {
		writeError(w, http.StatusInternalServerError, "catalogue_query_failed", "The catalogue could not be loaded.")
		return
	}

	offset := (page - 1) * pageSize
	const listSelect = `
		SELECT
			o.id::text,
			o.public_code,
			o.market_id::text,
			m.name,
			COALESCE(o.public_category_id::text, ''),
			COALESCE(pc.name, ''),
			o.classification_id::text,
			o.title,
			COALESCE(o.summary, ''),
			o.offer_type,
			COALESCE(o.price_amount::text, ''),
			COALESCE(o.currency_code, ''),
			COALESCE(o.quantity_total::text, ''),
			COALESCE(o.quantity_unit, ''),
			o.attributes,
			o.published_at,
			o.valid_until
	`
	rows, err := h.DB.QueryContext(
		r.Context(),
		listSelect+filter+`
			ORDER BY o.published_at DESC, o.id DESC
			LIMIT $4 OFFSET $5
		`,
		search, categoryID, marketID, pageSize, offset,
	)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "catalogue_query_failed", "The catalogue could not be loaded.")
		return
	}
	defer rows.Close()

	data := make([]OfferSummary, 0, pageSize)
	for rows.Next() {
		var item OfferSummary
		var categoryIDValue string
		var categoryName string
		var validUntil sql.NullTime
		var attributes []byte

		if err := rows.Scan(
			&item.ID,
			&item.Code,
			&item.MarketID,
			&item.MarketName,
			&categoryIDValue,
			&categoryName,
			&item.ClassificationID,
			&item.Title,
			&item.Summary,
			&item.OfferType,
			&item.PriceAmount,
			&item.CurrencyCode,
			&item.QuantityTotal,
			&item.QuantityUnit,
			&attributes,
			&item.PublishedAt,
			&validUntil,
		); err != nil {
			writeError(w, http.StatusInternalServerError, "catalogue_scan_failed", "A catalogue record could not be read.")
			return
		}

		item.PublicCategoryID = categoryIDValue
		item.PublicCategoryName = categoryName
		if validUntil.Valid {
			value := validUntil.Time
			item.ValidUntil = &value
		}
		if len(attributes) == 0 {
			item.Attributes = json.RawMessage(`{}`)
		} else if json.Valid(attributes) {
			item.Attributes = json.RawMessage(append([]byte(nil), attributes...))
		} else {
			writeError(w, http.StatusInternalServerError, "invalid_catalogue_data", "A catalogue record contains invalid attributes.")
			return
		}
		data = append(data, item)
	}
	if err := rows.Err(); err != nil {
		writeError(w, http.StatusInternalServerError, "catalogue_read_failed", "The catalogue could not be read completely.")
		return
	}

	totalPages := int64(0)
	if total > 0 {
		totalPages = (total + int64(pageSize) - 1) / int64(pageSize)
	}

	writeJSON(w, http.StatusOK, ListResponse{
		Data:       data,
		Page:       page,
		PageSize:   pageSize,
		Total:      total,
		TotalPages: totalPages,
	})
}

func (h Handler) GetOffer(w http.ResponseWriter, r *http.Request) {
	if h.DB == nil {
		writeError(w, http.StatusServiceUnavailable, "database_unavailable", "Catalogue storage is unavailable.")
		return
	}

	id := strings.TrimSpace(r.PathValue("id"))
	if id == "" || len(id) > 64 {
		writeError(w, http.StatusBadRequest, "invalid_offer_id", "A valid offer ID is required.")
		return
	}

	const query = `
		SELECT
			o.id::text,
			o.public_code,
			o.market_id::text,
			m.name,
			COALESCE(o.public_category_id::text, ''),
			COALESCE(pc.name, ''),
			o.classification_id::text,
			o.title,
			COALESCE(o.summary, ''),
			o.offer_type,
			COALESCE(o.price_amount::text, ''),
			COALESCE(o.currency_code, ''),
			COALESCE(o.quantity_total::text, ''),
			COALESCE(o.quantity_unit, ''),
			o.attributes,
			o.published_at,
			o.valid_until
		FROM offers o
		INNER JOIN markets m ON m.id = o.market_id
		LEFT JOIN public_categories pc ON pc.id = o.public_category_id
		WHERE o.id::text = $1
		  AND o.status = 'published'
		  AND m.status = 'active'
		  AND (o.valid_from IS NULL OR o.valid_from <= now())
		  AND (o.valid_until IS NULL OR o.valid_until > now())
	`

	var item OfferSummary
	var categoryIDValue string
	var categoryName string
	var validUntil sql.NullTime
	var attributes []byte

	err := h.DB.QueryRowContext(r.Context(), query, id).Scan(
		&item.ID,
		&item.Code,
		&item.MarketID,
		&item.MarketName,
		&categoryIDValue,
		&categoryName,
		&item.ClassificationID,
		&item.Title,
		&item.Summary,
		&item.OfferType,
		&item.PriceAmount,
		&item.CurrencyCode,
		&item.QuantityTotal,
		&item.QuantityUnit,
		&attributes,
		&item.PublishedAt,
		&validUntil,
	)

	if err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			writeError(w, http.StatusNotFound, "offer_not_found", "The published offer was not found.")
			return
		}
		writeError(w, http.StatusInternalServerError, "catalogue_query_failed", "The offer could not be loaded.")
		return
	}

	item.PublicCategoryID = categoryIDValue
	item.PublicCategoryName = categoryName
	if validUntil.Valid {
		value := validUntil.Time
		item.ValidUntil = &value
	}
	if len(attributes) == 0 {
		item.Attributes = json.RawMessage(`{}`)
	} else if json.Valid(attributes) {
		item.Attributes = json.RawMessage(append([]byte(nil), attributes...))
	} else {
		writeError(w, http.StatusInternalServerError, "invalid_catalogue_data", "The offer contains invalid attributes.")
		return
	}

	writeJSON(w, http.StatusOK, item)
}

func parsePositiveInt(raw string, fallback, maximum int) (int, error) {
	if raw == "" {
		return fallback, nil
	}
	value, err := strconv.Atoi(raw)
	if err != nil || value < 1 || value > maximum {
		return 0, errors.New("out of range")
	}
	return value, nil
}

package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/config"
	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/internal/database"
	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/internal/httpapi"
)

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	slog.SetDefault(logger)

	cfg, err := config.Load()
	if err != nil {
		logger.Error("invalid configuration", "error", err)
		os.Exit(1)
	}

	startupCtx, startupCancel := context.WithTimeout(
		context.Background(), 10*time.Second,
	)

	db, err := database.Open(startupCtx, database.Config{
		URL:             cfg.DatabaseURL,
		MaxOpenConns:    cfg.DBMaxOpenConns,
		MaxIdleConns:    cfg.DBMaxIdleConns,
		ConnMaxLifetime: cfg.DBConnMaxLifetime,
	})
	startupCancel()

	if err != nil {
		logger.Error("database initialization failed", "error", err)
		os.Exit(1)
	}

	if db != nil {
		defer db.Close()
	}

	server := &http.Server{
		Addr:              cfg.HTTPAddr,
		Handler:           httpapi.New(db),
		ReadHeaderTimeout: 5 * time.Second,
		IdleTimeout:       60 * time.Second,
	}

	serverErrors := make(chan error, 1)

	go func() {
		logger.Info("HTTP server starting", "address", cfg.HTTPAddr)
		serverErrors <- server.ListenAndServe()
	}()

	signalCtx, stop := signal.NotifyContext(
		context.Background(), os.Interrupt, syscall.SIGTERM,
	)
	defer stop()

	select {
	case err := <-serverErrors:
		if !errors.Is(err, http.ErrServerClosed) {
			logger.Error("HTTP server stopped unexpectedly", "error", err)
			os.Exit(1)
		}

	case <-signalCtx.Done():
		shutdownCtx, cancel := context.WithTimeout(
			context.Background(), 10*time.Second,
		)
		defer cancel()

		if err := server.Shutdown(shutdownCtx); err != nil {
			logger.Error("graceful shutdown failed", "error", err)
			_ = server.Close()
		}
	}
}

package httpapi

import (
	"context"
	"database/sql"
	"encoding/json"
	"net/http"
	"time"

	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/internal/catalogue"

func New(db *sql.DB) http.Handler {
	mux := http.NewServeMux()

	mux.HandleFunc("GET /health/live", func(w http.ResponseWriter, r *http.Request) {
		writeHealthJSON(w, http.StatusOK, map[string]string{
			"status": "alive",
		})
	})

	mux.HandleFunc("GET /health/ready", func(w http.ResponseWriter, r *http.Request) {
		if db == nil {
			writeHealthJSON(w, http.StatusServiceUnavailable, map[string]string{
				"status": "not_ready",
				"reason": "database_unavailable",
			})
			return
		}

		ctx, cancel := timeouts(r)
		defer cancel()

		if err := db.PingContext(ctx); err != nil {
			writeHealthJSON(w, http.StatusServiceUnavailable, map[string]string{
				"status": "not_ready",
				"reason": "database_unavailable",
			})
			return
		}

		writeHealthJSON(w, http.StatusOK, map[string]string{
			"status": "ready",
		})
	})

	catalogue.Handler{DB: db}.Register(mux)

	return securityHeaders(mux)
}

func timeouts(r *http.Request) (context.Context, context.CancelFunc) {
	return context.WithTimeout(r.Context(), 2*time.Second)
}

func writeHealthJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.Header().Set("X-Content-Type-Options", "nosniff")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func securityHeaders(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("X-Content-Type-Options", "nosniff")
		w.Header().Set("X-Frame-Options", "DENY")
		w.Header().Set("Referrer-Policy", "no-referrer")
		w.Header().Set("Cache-Control", "no-store")
		next.ServeHTTP(w, r)
	})
}

package config

import (
	"fmt"
	"os"
	"strconv"
	"time"
)

type Config struct {
	HTTPAddr          string
	DatabaseURL       string
	DBMaxOpenConns    int
	DBMaxIdleConns    int
	DBConnMaxLifetime time.Duration
}

func Load() (Config, error) {
	cfg := Config{
		HTTPAddr:          env("HTTP_ADDR", ":8080"),
		DatabaseURL:       os.Getenv("DATABASE_URL"),
		DBMaxOpenConns:    10,
		DBMaxIdleConns:    5,
		DBConnMaxLifetime: 5 * time.Minute,
	}

	var err error

	cfg.DBMaxOpenConns, err = intEnv("DB_MAX_OPEN_CONNS", 10)
	if err != nil {
		return Config{}, err
	}

	cfg.DBMaxIdleConns, err = intEnv("DB_MAX_IDLE_CONNS", 5)
	if err != nil {
		return Config{}, err
	}

	if cfg.DBMaxOpenConns < 1 ||
		cfg.DBMaxIdleConns < 0 ||
		cfg.DBMaxIdleConns > cfg.DBMaxOpenConns {
		return Config{}, fmt.Errorf("invalid database connection pool settings")
	}

	if value := os.Getenv("DB_CONN_MAX_LIFETIME"); value != "" {
		cfg.DBConnMaxLifetime, err = time.ParseDuration(value)
		if err != nil || cfg.DBConnMaxLifetime <= 0 {
			return Config{}, fmt.Errorf("invalid DB_CONN_MAX_LIFETIME")
		}
	}

	return cfg, nil
}

func env(key, fallback string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return fallback
}

func intEnv(key string, fallback int) (int, error) {
	value := os.Getenv(key)
	if value == "" {
		return fallback, nil
	}

	n, err := strconv.Atoi(value)
	if err != nil {
		return 0, fmt.Errorf("invalid %s: %w", key, err)
	}

	return n, nil
}
