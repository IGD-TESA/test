-- Additive, versioned and non-destructive.
--
-- Ownership boundaries:
-- * No account credentials or authoritative role database.
-- * No free-balance ledger or authoritative reservation ledger.
-- * No official property ownership register.
-- * No legal contract body owned by Contracts service.
-- * No service-delivery records owned by Services.
-- * No final settlement ledger owned by Settlement.
--
-- External identifiers are stored as references, not copied records.

);

-- Integrity hardening. Additive and transactional.
-- This file must never replace or modify an earlier migration.

-- 1. Specialized classification versioning.
ALTER TABLE specialized_classifications
    DROP CONSTRAINT IF EXISTS specialized_classifications_code_key;

CREATE UNIQUE INDEX IF NOT EXISTS uq_specialized_classifications_code_version
    ON specialized_classifications (code, version);

CREATE UNIQUE INDEX IF NOT EXISTS uq_specialized_classifications_one_active_code
    ON specialized_classifications (code)
    WHERE is_active = TRUE;

-- 2. Global category mappings.
-- The original UNIQUE constraint already protects non-NULL market_id rows.
-- This partial index closes PostgreSQL's NULL uniqueness gap.
CREATE UNIQUE INDEX IF NOT EXISTS uq_category_mappings_global
    ON category_mappings (public_category_id, specialized_classification_id)
    WHERE market_id IS NULL;

-- 3. Store the offer identity on each match.
ALTER TABLE order_matches
    ADD COLUMN IF NOT EXISTS offer_id UUID;

UPDATE order_matches AS om
SET offer_id = buy_order.offer_id
FROM orders AS buy_order
WHERE om.buy_order_id = buy_order.id
  AND om.offer_id IS NULL;

DO $$
BEGIN
    IF EXISTS (
        SELECT 1
        FROM order_matches AS om
        JOIN orders AS buy_order ON buy_order.id = om.buy_order_id
        JOIN orders AS sell_order ON sell_order.id = om.sell_order_id
        WHERE buy_order.offer_id <> sell_order.offer_id
           OR om.offer_id IS NULL
           OR om.offer_id <> buy_order.offer_id
    ) THEN
        RAISE EXCEPTION
            'Migration 000003 stopped: an order match references inconsistent offers';
    END IF;
END
$$;

ALTER TABLE order_matches
    ALTER COLUMN offer_id SET NOT NULL;

-- 4. Composite unique keys used by referential integrity constraints.
CREATE UNIQUE INDEX IF NOT EXISTS uq_orders_id_offer_id
    ON orders (id, offer_id);

CREATE UNIQUE INDEX IF NOT EXISTS uq_order_matches_identity
    ON order_matches (id, buy_order_id, sell_order_id, offer_id);

-- 5. Each side of a match must belong to the same offer recorded by the match.
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
        WHERE conrelid = 'order_matches'::regclass
          AND conname = 'fk_match_buy_order_offer'
    ) THEN
        ALTER TABLE order_matches
            ADD CONSTRAINT fk_match_buy_order_offer
            FOREIGN KEY (buy_order_id, offer_id)
            REFERENCES orders (id, offer_id)
            ON DELETE RESTRICT;
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
        WHERE conrelid = 'order_matches'::regclass
          AND conname = 'fk_match_sell_order_offer'
    ) THEN
        ALTER TABLE order_matches
            ADD CONSTRAINT fk_match_sell_order_offer
            FOREIGN KEY (sell_order_id, offer_id)
            REFERENCES orders (id, offer_id)
            ON DELETE RESTRICT;
    END IF;
END
$$;

-- 6. A trade must reference the exact match and exact two orders.
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint
        WHERE conrelid = 'trades'::regclass
          AND conname = 'fk_trade_match_order_pair'
    ) THEN
        ALTER TABLE trades
            ADD CONSTRAINT fk_trade_match_order_pair
            FOREIGN KEY (match_id, buy_order_id, sell_order_id, offer_id)
            REFERENCES order_matches (id, buy_order_id, sell_order_id, offer_id)
            ON DELETE RESTRICT;
    END IF;

        SELECT 1 FROM pg_constraint

    ) THEN

package database

import (
	"context"
	"database/sql"
	"fmt"
	"time"

	_ "github.com/jackc/pgx/v5/stdlib"
)

type Config struct {
	URL             string
	MaxOpenConns    int
	MaxIdleConns    int
	ConnMaxLifetime time.Duration
}

func Open(ctx context.Context, cfg Config) (*sql.DB, error) {
	if cfg.URL == "" {
		return nil, nil
	}

	db, err := sql.Open("pgx", cfg.URL)
	if err != nil {
		return nil, fmt.Errorf("open database: %w", err)
	}

	if cfg.MaxOpenConns > 0 {
		db.SetMaxOpenConns(cfg.MaxOpenConns)
	}
	if cfg.MaxIdleConns >= 0 {
		db.SetMaxIdleConns(cfg.MaxIdleConns)
	}
	if cfg.ConnMaxLifetime > 0 {
		db.SetConnMaxLifetime(cfg.ConnMaxLifetime)
	}

	pingCtx, cancel := context.WithTimeout(ctx, 5*time.Second)
	defer cancel()

	if err := db.PingContext(pingCtx); err != nil {
		_ = db.Close()
		return nil, fmt.Errorf("connect to database: %w", err)
	}

	return db, nil
}

package database

import (
	"context"
	"crypto/sha256"
	"database/sql"
	"encoding/hex"
	"fmt"
	"io/fs"
	"sort"
	"strings"
	"time"

	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/migrations"
)

const migrationLockID int64 = 5270645408566230100

func Migrate(ctx context.Context, db *sql.DB) error {
	if db == nil {
		return fmt.Errorf("database is not configured")
	}

	conn, err := db.Conn(ctx)
	if err != nil {
		return fmt.Errorf("acquire migration connection: %w", err)
	}
	defer conn.Close()

	if _, err := conn.ExecContext(
		ctx,
		"SELECT pg_advisory_lock($1)",
		migrationLockID,
	); err != nil {
		return fmt.Errorf("acquire migration advisory lock: %w", err)
	}

	defer func() {
		unlockCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		_, _ = conn.ExecContext(
			unlockCtx,
			"SELECT pg_advisory_unlock($1)",
			migrationLockID,
		)
	}()

	_, err = conn.ExecContext(ctx, `
		CREATE TABLE IF NOT EXISTS schema_migrations (
			version TEXT PRIMARY KEY,
			checksum TEXT,
			applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
		)
	`)
	if err != nil {
		return fmt.Errorf("initialize migration tracking: %w", err)
	}

	_, err = conn.ExecContext(ctx, `
		ALTER TABLE schema_migrations
		ADD COLUMN IF NOT EXISTS checksum TEXT
	`)
	if err != nil {
		return fmt.Errorf("upgrade migration tracking: %w", err)
	}

	files, err := fs.Glob(migrations.Files, "*.sql")
	if err != nil {
		return fmt.Errorf("list migration files: %w", err)
	}
	sort.Strings(files)

	for _, file := range files {
		if strings.Contains(file, "..") {
			return fmt.Errorf("invalid migration path: %s", file)
		}

		content, err := migrations.Files.ReadFile(file)
		if err != nil {
			return fmt.Errorf("read migration %s: %w", file, err)
		}

		version := strings.TrimSuffix(file, ".sql")
		sum := sha256.Sum256(content)
		checksum := hex.EncodeToString(sum[:])

		var stored sql.NullString
		err = conn.QueryRowContext(
			ctx,
			`SELECT checksum FROM schema_migrations WHERE version = $1`,
			version,
		).Scan(&stored)

		if err == nil {
			if !stored.Valid || stored.String == "" {
				// Existing installations created before checksum support:
				// adopt the currently deployed migration as the baseline.
				_, err = conn.ExecContext(
					ctx,
					`UPDATE schema_migrations
					 SET checksum = $1
					 WHERE version = $2 AND checksum IS NULL`,
					checksum,
					version,
				)
				if err != nil {
					return fmt.Errorf("adopt legacy checksum for %s: %w", version, err)
				}
				continue
			}

			if stored.String != checksum {
				return fmt.Errorf(
					"migration checksum mismatch for %s: applied migration file was modified; restore the original file or create a new migration",
					version,
				)
			}
			continue
		}

		if err != sql.ErrNoRows {
			return fmt.Errorf("check migration %s: %w", version, err)
		}

		tx, err := conn.BeginTx(ctx, nil)
		if err != nil {
			return fmt.Errorf("begin migration %s: %w", version, err)
		}

		if _, err = tx.ExecContext(ctx, string(content)); err == nil {
			_, err = tx.ExecContext(
				ctx,
				`INSERT INTO schema_migrations (version, checksum)
				 VALUES ($1, $2)`,
				version,
				checksum,
			)
		}

		if err != nil {
			_ = tx.Rollback()
			return fmt.Errorf("apply migration %s: %w", version, err)
		}

		if err := tx.Commit(); err != nil {
			return fmt.Errorf("commit migration %s: %w", version, err)
		}
	}

	return nil
}

package main

import (
	"context"
	"log"
	"os"
	"time"

	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/config"
	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/internal/database"
)

func main() {
	cfg, err := config.Load()
	if err != nil {
		log.Fatalf("invalid configuration: %v", err)
	}

	if cfg.DatabaseURL == "" {
		log.Fatal("DATABASE_URL is required for migration execution")
	}

	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()

	db, err := database.Open(ctx, database.Config{
		URL:             cfg.DatabaseURL,
		MaxOpenConns:    cfg.DBMaxOpenConns,
		MaxIdleConns:    cfg.DBMaxIdleConns,
		ConnMaxLifetime: cfg.DBConnMaxLifetime,
	})
	if err != nil {
		log.Fatalf("database connection failed: %v", err)
	}
	defer db.Close()

	if err := database.Migrate(ctx, db); err != nil {
		log.Fatalf("migration failed: %v", err)
	}

	log.Print("Database migrations completed successfully.")
	os.Exit(0)
}

package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/config"
	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/internal/database"
	"github.com/igd-tesa/real-asset-management-platform/general-market-backend/internal/httpapi"
)

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	slog.SetDefault(logger)

	cfg, err := config.Load()
	if err != nil {
		logger.Error("invalid configuration", "error", err)
		os.Exit(1)
	}

	startupCtx, startupCancel := context.WithTimeout(
		context.Background(), 10*time.Second,
	)

	db, err := database.Open(startupCtx, database.Config{
		URL:             cfg.DatabaseURL,
		MaxOpenConns:    cfg.DBMaxOpenConns,
		MaxIdleConns:    cfg.DBMaxIdleConns,
		ConnMaxLifetime: cfg.DBConnMaxLifetime,
	})
	startupCancel()

	if err != nil {
		logger.Error("database initialization failed", "error", err)
		os.Exit(1)
	}

	if db != nil {
		defer db.Close()
	}

	server := &http.Server{
		Addr:              cfg.HTTPAddr,
		Handler:           httpapi.New(db),
		ReadHeaderTimeout: 5 * time.Second,
		IdleTimeout:       60 * time.Second,
	}

	serverErrors := make(chan error, 1)

	go func() {
		logger.Info("HTTP server starting", "address", cfg.HTTPAddr)
		serverErrors <- server.ListenAndServe()
	}()

	signalCtx, stop := signal.NotifyContext(
		context.Background(), os.Interrupt, syscall.SIGTERM,
	)
	defer stop()

	select {
	case err := <-serverErrors:
		if !errors.Is(err, http.ErrServerClosed) {
			logger.Error("HTTP server stopped unexpectedly", "error", err)
			os.Exit(1)
		}

	case <-signalCtx.Done():
		shutdownCtx, cancel := context.WithTimeout(
			context.Background(), 10*time.Second,
		)
		defer cancel()

		if err := server.Shutdown(shutdownCtx); err != nil {
			logger.Error("graceful shutdown failed", "error", err)
			_ = server.Close()
		}
	}
}

package config

import (
	"fmt"
	"os"
	"strconv"
	"time"
)

type Config struct {
	HTTPAddr          string
	DatabaseURL       string
	DBMaxOpenConns    int
	DBMaxIdleConns    int
	DBConnMaxLifetime time.Duration
}

func Load() (Config, error) {
	cfg := Config{
		HTTPAddr:          env("HTTP_ADDR", ":8080"),
		DatabaseURL:       os.Getenv("DATABASE_URL"),
		DBMaxOpenConns:    10,
		DBMaxIdleConns:    5,
		DBConnMaxLifetime: 5 * time.Minute,
	}

	var err error

	cfg.DBMaxOpenConns, err = intEnv("DB_MAX_OPEN_CONNS", 10)
	if err != nil {
		return Config{}, err
	}

	cfg.DBMaxIdleConns, err = intEnv("DB_MAX_IDLE_CONNS", 5)
	if err != nil {
		return Config{}, err
	}

	if cfg.DBMaxOpenConns < 1 ||
		cfg.DBMaxIdleConns < 0 ||
		cfg.DBMaxIdleConns > cfg.DBMaxOpenConns {
		return Config{}, fmt.Errorf("invalid database connection pool settings")
	}

	if value := os.Getenv("DB_CONN_MAX_LIFETIME"); value != "" {
		cfg.DBConnMaxLifetime, err = time.ParseDuration(value)
		if err != nil || cfg.DBConnMaxLifetime <= 0 {
			return Config{}, fmt.Errorf("invalid DB_CONN_MAX_LIFETIME")
		}
	}

	return cfg, nil
}

func env(key, fallback string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return fallback
}

func intEnv(key string, fallback int) (int, error) {
	value := os.Getenv(key)
	if value == "" {
		return fallback, nil
	}

	n, err := strconv.Atoi(value)
	if err != nil {
		return 0, fmt.Errorf("invalid %s: %w", key, err)
	}

	return n, nil
}
