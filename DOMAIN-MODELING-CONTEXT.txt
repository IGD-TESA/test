package domain

import (
	"errors"
	"fmt"
	"strings"
	"time"
)

var (
	ErrInvalidMarketID       = errors.New("invalid market id")
	ErrInvalidMarketCode     = errors.New("invalid market code")
	ErrInvalidMarketName     = errors.New("invalid market name")
	ErrInvalidMarketTimezone = errors.New("invalid market timezone")
	ErrInvalidTimestamp      = errors.New("invalid timestamp")
	ErrInvalidMarketState    = errors.New("invalid market state")
	ErrMarketClosed          = errors.New("market is closed")
)

type MarketID string

type MarketStatus string

const (
	MarketStatusDraft     MarketStatus = "draft"
	MarketStatusActive    MarketStatus = "active"
	MarketStatusSuspended MarketStatus = "suspended"
	MarketStatusClosed    MarketStatus = "closed"
)

func (s MarketStatus) IsValid() bool {
	switch s {
	case MarketStatusDraft,
		MarketStatusActive,
		MarketStatusSuspended,
		MarketStatusClosed:
		return true
	default:
		return false
	}
}

type CreateMarketInput struct {
	ID          MarketID
	Code        string
	Name        string
	Description string
	Timezone    string
}

type Market struct {
	id          MarketID
	code        string
	name        string
	description string
	timezone    string

	status    MarketStatus
	version   uint64
	createdAt time.Time
	updatedAt time.Time
}

func NewMarket(input CreateMarketInput, now time.Time) (*Market, error) {
	if strings.TrimSpace(string(input.ID)) == "" {
		return nil, ErrInvalidMarketID
	}

	code := normalizeMarketCode(input.Code)
	if !isValidMarketCode(code) {
		return nil, ErrInvalidMarketCode
	}

	name := strings.TrimSpace(input.Name)
	if name == "" || len([]rune(name)) > 160 {
		return nil, ErrInvalidMarketName
	}

	if now.IsZero() {
		return nil, ErrInvalidTimestamp
	}

	timezone := strings.TrimSpace(input.Timezone)
	if timezone == "" {
		timezone = "UTC"
	}

	if _, err := time.LoadLocation(timezone); err != nil {
		return nil, fmt.Errorf("%w: %s", ErrInvalidMarketTimezone, timezone)
	}

	return &Market{
		id:          input.ID,
		code:        code,
		name:        name,
		description: strings.TrimSpace(input.Description),
		timezone:    timezone,
		status:      MarketStatusDraft,
		version:     1,
		createdAt:   now.UTC(),
		updatedAt:   now.UTC(),
	}, nil
}

func normalizeMarketCode(value string) string {
	return strings.ToUpper(strings.TrimSpace(value))
}

func isValidMarketCode(value string) bool {
	if len(value) < 2 || len(value) > 32 {
		return false
	}

	for _, r := range value {
		valid := (r >= 'A' && r <= 'Z') ||
			(r >= '0' && r <= '9') ||
			r == '-' ||
			r == '_'

		if !valid {
			return false
		}
	}

	return true
}

func (m *Market) UpdateDetails(
	name string,
	description string,
	timezone string,
	now time.Time,
) error {
	if m.status == MarketStatusClosed {
		return ErrMarketClosed
	}

	name = strings.TrimSpace(name)
	if name == "" || len([]rune(name)) > 160 {
		return ErrInvalidMarketName
	}

	timezone = strings.TrimSpace(timezone)
	if timezone == "" {
		timezone = "UTC"
	}

	if _, err := time.LoadLocation(timezone); err != nil {
		return fmt.Errorf("%w: %s", ErrInvalidMarketTimezone, timezone)
	}

	if err := m.validateTimestamp(now); err != nil {
		return err
	}

	m.name = name
	m.description = strings.TrimSpace(description)
	m.timezone = timezone
	m.touch(now)

	return nil
}

func (m *Market) Activate(now time.Time) error {
	switch m.status {
	case MarketStatusDraft, MarketStatusSuspended:
		if err := m.validateTimestamp(now); err != nil {
			return err
		}

		m.status = MarketStatusActive
		m.touch(now)
		return nil

	case MarketStatusActive:
		return ErrInvalidMarketState

	case MarketStatusClosed:
		return ErrMarketClosed

	default:
		return ErrInvalidMarketState
	}
}

func (m *Market) Suspend(now time.Time) error {
	if m.status != MarketStatusActive {
		return ErrInvalidMarketState
	}

	if err := m.validateTimestamp(now); err != nil {
		return err
	}

	m.status = MarketStatusSuspended
	m.touch(now)

	return nil
}

func (m *Market) Close(now time.Time) error {
	if m.status == MarketStatusClosed {
		return ErrInvalidMarketState
	}

	if err := m.validateTimestamp(now); err != nil {
		return err
	}

	m.status = MarketStatusClosed
	m.touch(now)

	return nil
}

func (m *Market) IsOperational() bool {
	return m.status == MarketStatusActive
}

func (m *Market) validateTimestamp(now time.Time) error {
	if now.IsZero() {
		return ErrInvalidTimestamp
	}

	if now.Before(m.updatedAt) {
		return ErrInvalidTimestamp
	}

	return nil
}

func (m *Market) touch(now time.Time) {
	m.updatedAt = now.UTC()
	m.version++
}

func NewCategory(input CreateCategoryInput, now time.Time) (*Category, error) {

		return nil, ErrInvalidCategoryID
	}

	if !isValidCategoryCode(code) {
		return nil, ErrInvalidCategoryCode

	if name == "" || len([]rune(name)) > 160 {
		return nil, ErrInvalidCategoryName
	}

	if !input.Kind.IsValid() {
		return nil, ErrInvalidCategoryKind
	}

	if input.ParentID != "" && input.ParentID == input.ID {
		return nil, ErrInvalidParentCategory
	}

	if now.IsZero() {
		return nil, ErrInvalidTimestamp
	}

	return &Category{
		id:        input.ID,
		code:      code,
		name:      name,
		kind:      input.Kind,
		parentID:  input.ParentID,
		sortOrder: input.SortOrder,
		status:    CategoryStatusDraft,
		version:   1,
		createdAt: now.UTC(),
		updatedAt: now.UTC(),
	}, nil
}

package domain

import (
	"errors"
	"strings"
	"time"
)

var (
	ErrInvalidCategoryMappingID = errors.New("invalid category mapping id")
	ErrInvalidCategoryMapping   = errors.New("invalid category mapping")
	ErrCategoryMappingArchived  = errors.New("category mapping is archived")
	ErrCategoryMappingState     = errors.New("invalid category mapping state")
	ErrCategoryKindsMismatch    = errors.New("category kinds do not match mapping direction")
	ErrCategoryNotActive        = errors.New("both categories must be active")
)

type CategoryMappingID string

type CategoryMappingStatus string

const (
	CategoryMappingStatusDraft    CategoryMappingStatus = "draft"
	CategoryMappingStatusActive   CategoryMappingStatus = "active"
	CategoryMappingStatusArchived CategoryMappingStatus = "archived"
)

type CreateCategoryMappingInput struct {
	ID                    CategoryMappingID
	PublicCategoryID      CategoryID
	SpecializedCategoryID CategoryID
	MarketID              string
	Description           string
}

type CategoryMapping struct {
	id                    CategoryMappingID
	publicCategoryID      CategoryID
	specializedCategoryID CategoryID
	marketID              string
	description           string

	status    CategoryMappingStatus
	version   uint64
	createdAt time.Time
	updatedAt time.Time
}

func NewCategoryMapping(
	input CreateCategoryMappingInput,
	now time.Time,
) (*CategoryMapping, error) {
	if strings.TrimSpace(string(input.ID)) == "" {
		return nil, ErrInvalidCategoryMappingID
	}

	if strings.TrimSpace(string(input.PublicCategoryID)) == "" ||
		strings.TrimSpace(string(input.SpecializedCategoryID)) == "" ||
		input.PublicCategoryID == input.SpecializedCategoryID {
		return nil, ErrInvalidCategoryMapping
	}

	if now.IsZero() {
		return nil, ErrInvalidTimestamp
	}

	return &CategoryMapping{
		id:                    input.ID,
		publicCategoryID:      input.PublicCategoryID,
		specializedCategoryID: input.SpecializedCategoryID,
		marketID:              strings.TrimSpace(input.MarketID),
		description:           strings.TrimSpace(input.Description),
		status:                CategoryMappingStatusDraft,
		version:               1,
		createdAt:             now.UTC(),
		updatedAt:             now.UTC(),
	}, nil
}

func (m *CategoryMapping) ID() CategoryMappingID {
	return m.id
}

func (m *CategoryMapping) PublicCategoryID() CategoryID {
	return m.publicCategoryID
}

func (m *CategoryMapping) SpecializedCategoryID() CategoryID {
	return m.specializedCategoryID
}

func (m *CategoryMapping) MarketID() string {
	return m.marketID
}

func (m *CategoryMapping) Description() string {
	return m.description
}

func (m *CategoryMapping) Status() CategoryMappingStatus {
	return m.status
}

func (m *CategoryMapping) Version() uint64 {
	return m.version
}

func (m *CategoryMapping) CreatedAt() time.Time {
	return m.createdAt
}

func (m *CategoryMapping) UpdatedAt() time.Time {
	return m.updatedAt
}

func (m *CategoryMapping) Activate(
	publicCategory *Category,
	specializedCategory *Category,
	now time.Time,
) error {
	if m.status == CategoryMappingStatusArchived {
		return ErrCategoryMappingArchived
	}

	if m.status != CategoryMappingStatusDraft {
		return ErrCategoryMappingState
	}

	if publicCategory == nil || specializedCategory == nil {
		return ErrInvalidCategoryMapping
	}

	if publicCategory.ID() != m.publicCategoryID ||
		specializedCategory.ID() != m.specializedCategoryID {
		return ErrInvalidCategoryMapping
	}

	if publicCategory.Kind() != CategoryKindGeneral ||
		specializedCategory.Kind() != CategoryKindSpecialized {
		return ErrCategoryKindsMismatch
	}

	if !publicCategory.IsSelectable() || !specializedCategory.IsSelectable() {
		return ErrCategoryNotActive
	}

	if now.IsZero() || now.Before(m.updatedAt) {
		return ErrInvalidTimestamp
	}

	m.status = CategoryMappingStatusActive
	m.touch(now)

	return nil
}

func (m *CategoryMapping) Archive(now time.Time) error {
	if m.status == CategoryMappingStatusArchived {
		return ErrCategoryMappingState
	}

	if now.IsZero() || now.Before(m.updatedAt) {
		return ErrInvalidTimestamp
	}

	m.status = CategoryMappingStatusArchived
	m.touch(now)

	return nil
}

func (m *CategoryMapping) IsUsable() bool {
	return m.status == CategoryMappingStatusActive
}

func (m *CategoryMapping) touch(now time.Time) {
	m.updatedAt = now.UTC()
	m.version++
}
